cmake_minimum_required(VERSION 3.8)
project(mzj_control)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(mzj_interfaces REQUIRED)
find_package(mzj_model REQUIRED)
add_executable(${PROJECT_NAME}_node src/node.cpp)
target_include_directories(${PROJECT_NAME}_node PRIVATE include)
ament_target_dependencies(${PROJECT_NAME}_node rclcpp geometry_msgs nav_msgs mzj_interfaces mzj_model)
install(TARGETS ${PROJECT_NAME}_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include OPTIONAL)

ament_package()
